        function typeLetter() {
            const container = document.getElementById('letterContent');
            if (!container || letterTyped) return;
            letterTyped = true;
            let lineIndex = 0;
            let charIdx = 0;
            let currentP = null;

            function typeNextChar() {
                if (lineIndex >= letterLines.length) return;
                if (!currentP) {
                    currentP = document.createElement('p');
                    // Tiny per-line tilt and vertical drift, like a hand that doesn't write
                    // perfectly level on every line of a real letter.
                    const tilt = (Math.random() - 0.5) * 2.4;
                    const lift = (Math.random() - 0.5) * 6;
                    currentP.style.transform = `rotate(${tilt.toFixed(2)}deg) translateY(${lift.toFixed(1)}px)`;
                    const cursor = document.createElement('span');
                    cursor.className = 'letter-typing-cursor';
                    currentP.appendChild(document.createTextNode(''));
                    currentP.appendChild(cursor);
                    container.appendChild(currentP);
                }
                const line = letterLines[lineIndex];
                if (charIdx < line.length) {
                    currentP.firstChild.textContent += line.charAt(charIdx);
                    charIdx++;
                    // Small natural variance in typing speed, plus a tiny extra pause after
                    // punctuation, so it reads like a hand actually writing rather than a
                    // perfectly even machine tick.
                    const ch = line.charAt(charIdx - 1);
                    const pause = /[.,!—]/.test(ch) ? 140 : (35 + Math.random() * 45);
                    letterTypeTimer = setTimeout(typeNextChar, pause);
                } else {
                    // Line finished — remove its blinking cursor and move to the next line
                    const cursor = currentP.querySelector('.letter-typing-cursor');
                    if (cursor) cursor.remove();
                    currentP = null;
                    charIdx = 0;
                    lineIndex++;
                    letterTypeTimer = setTimeout(typeNextChar, 260);
                }
            }
            typeNextChar();
        }

        // Updated countdown function
        function updateCountdown() {
            const now = new Date();
            
            const isBirthdayToday = isBirthday(now);
            
            // Update message — kept generic on purpose so the countdown screen doesn't reveal
            // this is a birthday surprise before the reveal happens
            document.getElementById('countdownMessage').innerHTML = 
                `⏰ Something is coming soon...`;
            
            if (isBirthdayToday) {
                // It's birthday!
                if (!countdownFinished) {
                    countdownFinished = true;
                    countdownAudio.pause();
                    countdownAudio.currentTime = 0;
                    
                    // Show greeting screen
                    document.getElementById('countdownScreen').classList.add('hide');
                    document.getElementById('greetingScreen').classList.add('show');
                    
                    if (!confettiShown) {
                        confettiShown = true;
                        confetti({ particleCount: 100, spread: 70 });
                    }
                    
                    if (charIndex === 0) {
                        typeGreeting();
                    }
                    
                    // Try to play birthday audio
                    if (!birthdayAudioPlayed) {
                        playBirthdayAudio();
                    }
                    
                    scheduleAutoReset();
                }
                return;
            }
            
            // Not birthday - show countdown
            countdownFinished = false;
            document.getElementById('greetingScreen').classList.remove('show');
            document.getElementById('countdownScreen').classList.remove('hide');
            
            const currentYear = now.getFullYear();
            let birthday = new Date(currentYear, CONFIG.month, CONFIG.day, CONFIG.hour, CONFIG.minute, 0);
            
            if (now > birthday) {
                birthday = new Date(currentYear + 1, CONFIG.month, CONFIG.day, CONFIG.hour, CONFIG.minute, 0);
            }
            
            const diff = birthday - now;
            // Round up to whole seconds so the on-screen "Secs" counter and the audio
            // trigger below both agree on exactly when "10 seconds left" happens —
            // using raw diff (ms) for one and floor(diff/1000) for the other let the
            // two drift apart by up to a second because setInterval doesn't fire at
            // perfectly even 1000ms intervals.
            const totalSecondsLeft = Math.ceil(diff / 1000);
            
            if (diff > 0) {
                const days = Math.floor(totalSecondsLeft / (60 * 60 * 24));
                const hours = Math.floor((totalSecondsLeft / (60 * 60)) % 24);
                const mins = Math.floor((totalSecondsLeft / 60) % 60);
                const secs = totalSecondsLeft % 60;
                
                document.getElementById('days').textContent = String(days).padStart(2, '0');
                document.getElementById('hours').textContent = String(hours).padStart(2, '0');
                document.getElementById('mins').textContent = String(mins).padStart(2, '0');
                document.getElementById('secs').textContent = String(secs).padStart(2, '0');
            }
            
            // Play countdown audio in last 10 seconds — now driven by the same
            // totalSecondsLeft value shown on screen, so it fires exactly when the
            // counter reads "10", not a second early or late.
            if (totalSecondsLeft <= 10 && diff > 0 && !countdownAudioPlayed && audioAllowed) {
                countdownAudio.play()
                    .then(() => { countdownAudioPlayed = true; })
                    .catch(error => {});
            }
        }

        // Grab the on-screen center of each currently-lit candle before renderCake() wipes and
        // rebuilds the candle list, so the smoke can rise from exactly where each flame was.
        function candleCenters(fromIndex, toIndex) {
            const spans = document.querySelectorAll('#candles .candle');
            const centers = [];
            for (let i = fromIndex; i < toIndex; i++) {
                const el = spans[i];
                if (!el) continue;
                const r = el.getBoundingClientRect();
                centers.push({ x: r.left + r.width / 2, y: r.top + r.height * 0.25 });
            }
            return centers;
        }

        window.blowCandle = function(index) {
            if (cakeCut || candlesBlown === totalCandles || index < candlesBlown) return;
            const centers = candleCenters(index, index + 1);
            candlesBlown = index + 1; 
            saveState(); 
            renderCake(); 
            playBlowPuff();
            vibrate(25);
            confetti({ particleCount:5, spread:30, colors:['#aaaaaa'] });
            centers.forEach(c => spawnSmokeAt(c.x, c.y));

        function blowMultipleCandles(count) {
            if (cakeCut || candlesBlown === totalCandles) return;
            const newBlown = Math.min(candlesBlown + count, totalCandles);
            if (newBlown > candlesBlown) { 
                const centers = candleCenters(candlesBlown, newBlown);
                candlesBlown = newBlown; 
                saveState(); 
                renderCake(); 
                playBlowPuff();
                vibrate(30);
                confetti({ particleCount:8 * count, spread:40, colors:['#aaaaaa'] }); 
                centers.forEach(c => spawnSmokeAt(c.x, c.y));
            }
        }

        // No button anymore — the browser's own permission popup is triggered automatically
        // (see requestMicAuto/showWishMoment). If the person denies it or the device has no
        // mic, the candles simply stay tap-to-blow, which already works.
        function startMicrophone() {
            if (cakeCut || candlesBlown === totalCandles) return;
            if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
                const hint = document.getElementById('micHint');
                hint.textContent = "⚠️ Microphone isn't supported on this device — tap to blow out the candles";
                hint.classList.add('denied');
                return;
            }
            
            navigator.mediaDevices.getUserMedia({ audio: true })
                .then(stream => {
                    audioStream = stream;
                    audioContext = new (window.AudioContext || window.webkitAudioContext)();
                    analyser = audioContext.createAnalyser();
                    const source = audioContext.createMediaStreamSource(stream);
                    source.connect(analyser);
                    analyser.fftSize = 256;
                    const dataArray = new Uint8Array(analyser.frequencyBinCount);

                    micActive = true;
                    micDenied = false;
                    const micHint = document.getElementById('micHint');
                    micHint.classList.add('listening');
                    micHint.classList.remove('denied');
                    micHint.textContent = '🎤 Now blow...';
                    
                    if (blowInterval) clearInterval(blowInterval);
                    blowInterval = setInterval(() => {
                        if (cakeCut || candlesBlown === totalCandles) { 
                            stopMicrophone(); 
                            return; 
                        }
                        analyser.getByteFrequencyData(dataArray);
                        let sum = 0;
                        for (let i = 0; i < dataArray.length; i++) sum += dataArray[i];
                        const avg = sum / dataArray.length;
                        if (avg > 50) { 
                            blowMultipleCandles(2 + Math.floor(Math.random() * 2)); 
                        }
                    }, 150);
                })
                .catch(() => {
                    micDenied = true;
                    const hint = document.getElementById('micHint');
                    hint.textContent = "⚠️ Mic permission wasn't granted — just tap to blow out the candles";
                    hint.classList.add('denied');
                });
        }

        function stopMicrophone() {
            if (blowInterval) { 
                clearInterval(blowInterval); 
                blowInterval = null; 
            }
            if (audioStream) { 
                audioStream.getTracks().forEach(t => t.stop()); 
                audioStream = null; 
            }
            if (audioContext && audioContext.state !== 'closed') { 
                audioContext.close(); 
                audioContext = null; 
            }
            micActive = false;
            const micHint = document.getElementById('micHint');
            if (micHint) {
                micHint.classList.remove('listening');
                if (candlesBlown < totalCandles && !cakeCut && !micDenied) {
                    micHint.textContent = '🎤 Blow into your mic to blow out the candles, or tap them';
                }
            }
        }

        // Animate the cut wedge actually leaving the cake and landing on the plate beside it
        function serveSliceOntoPlate() {
            const slice = document.getElementById('plateSlice');
            const cakeZone = document.getElementById('cakeEmoji');
            const plateWrap = document.querySelector('.plate-wrap');
            if (!slice || !cakeZone || !plateWrap) return;

            const cakeRect = cakeZone.getBoundingClientRect();
            const plateRect = plateWrap.getBoundingClientRect();

            // Start exactly where the wedge sits on the cake — wherever the user actually cut it
            const wedgePoint = angleToCakePoint(cutAngle, 0.72);
            const startX = cakeRect.left + (wedgePoint.x / 400) * cakeRect.width;
            const startY = cakeRect.top + (wedgePoint.y / 400) * cakeRect.height;
            // Land centered on the plate
            const endX = plateRect.left + plateRect.width / 2;
            const endY = plateRect.top + plateRect.height * 0.4;

            slice.style.position = 'fixed';
            slice.style.left = startX + 'px';
            slice.style.top = startY + 'px';
            slice.style.transform = 'translate(-50%,-50%) rotate(0deg) scale(1)';
            slice.style.opacity = '1';

            gsap.timeline()
                .to(slice, {
                    left: (startX + endX) / 2,
                    top: Math.min(startY, endY) - 60,
                    rotation: 25,
                    duration: 0.35,
                    ease: 'power1.out'
                })
                .to(slice, {
                    left: endX,
                    top: endY,
                    rotation: 8,
                    scale: 0.85,
                    duration: 0.35,
                    ease: 'power1.in',
                    onComplete: () => { vibrate(20); }
                })
                .to(slice, { scale: 0.95, duration: 0.12, yoyo: true, repeat: 1 });
        }

        // Drag-to-cut: hold the knife and swipe across the cake instead of tapping a button.
        // The cut is fully freeform — it follows wherever the user actually drags, in whatever
        // direction they choose, and the slice that gets served is the one they carved out.
        const CUT_DRAG_THRESHOLD = 110; // px of drag needed to complete the cut (any direction)
        const CAKE_CX = 200, CAKE_CY = 195, CAKE_RX = 175, CAKE_RY = 160; // matches the cake ellipse in the SVG
        const WEDGE_HALF_ANGLE = 26 * Math.PI / 180; // how wide the served slice is, either side of the cut
        let cutDragActive = false;
        let cutStartX = 0;
        let cutStartY = 0;
        let cutAngle = 0; // radians — direction of the cut the user is currently dragging
        let lastLineLength = 330;

        // Point on the cake's rim in the direction of `angle`, pulled in slightly (scale<1) so the
        // cut line reads as sitting on the cake surface rather than exactly on its outer edge.
        function angleToCakePoint(angle, scale) {
            return {
                x: CAKE_CX + Math.cos(angle) * CAKE_RX * scale,
                y: CAKE_CY + Math.sin(angle) * CAKE_RY * scale
            };
        }

        // Redraw the cut line so it runs straight through the cake along whatever angle the user
        // is currently dragging, instead of always being the fixed horizontal line it used to be.
        function updateSliceLineAngle(angle) {
            const p1 = angleToCakePoint(angle + Math.PI, 0.89);
            const p2 = angleToCakePoint(angle, 0.89);
            const length = Math.hypot(p2.x - p1.x, p2.y - p1.y);
            lastLineLength = length;
            const line = document.getElementById('sliceLine');
            const shadow = document.getElementById('sliceShadow');
            [line, shadow].forEach(el => {
                if (!el) return;
                el.setAttribute('x1', p1.x); el.setAttribute('y1', p1.y);
                el.setAttribute('x2', p2.x); el.setAttribute('y2', p2.y);
                el.style.strokeDasharray = String(length);
            });
            return length;
        }

        // Build the pie-slice path for whatever portion of the cake the user chose to cut,
        // centered on the angle they dragged toward.
        function buildWedgePaths(angle) {
            const p1 = angleToCakePoint(angle - WEDGE_HALF_ANGLE, 1);
            const p3 = angleToCakePoint(angle + WEDGE_HALF_ANGLE, 1);
            const fill = `M ${CAKE_CX} ${CAKE_CY} L ${p1.x} ${p1.y} A ${CAKE_RX} ${CAKE_RY} 0 0 1 ${p3.x} ${p3.y} Z`;
            const stroke = `M ${p1.x} ${p1.y} A ${CAKE_RX} ${CAKE_RY} 0 0 1 ${p3.x} ${p3.y}`;
            return { fill, stroke };
        }

        function applyWedgeAtAngle(angle) {
            const paths = buildWedgePaths(angle);
            const fillEl = document.getElementById('wedgeFill');
            const strokeEl = document.getElementById('wedgeStroke');
            if (fillEl) fillEl.setAttribute('d', paths.fill);
            if (strokeEl) strokeEl.setAttribute('d', paths.stroke);
        }

        function resetSliceLine() {
            const length = updateSliceLineAngle(0);
            const line = document.getElementById('sliceLine');
            const shadow = document.getElementById('sliceShadow');
            if (line) line.style.strokeDashoffset = String(length);
            if (shadow) shadow.style.strokeDashoffset = String(length);
        }

        function canDragCut() {
            return !cakeCut && candlesBlown === totalCandles;
        }

        function updateKnifePosition(clientX, clientY) {
            const zone = document.getElementById('cakeEmoji');
            const knife = document.getElementById('knifeCursor');
            if (!zone || !knife) return;
            const rect = zone.getBoundingClientRect();
            knife.style.left = (clientX - rect.left) + 'px';
            knife.style.top = (clientY - rect.top) + 'px';
        }

        // Angle (in the cake's own coordinate space) from the cake center out to wherever the
        // pointer currently is — this is what lets the user cut in literally any direction.
        function pointerToCakeAngle(clientX, clientY) {
            const zone = document.getElementById('cakeEmoji');
            const rect = zone.getBoundingClientRect();
            const scaleX = 400 / rect.width;
            const scaleY = 400 / rect.height;
            const svgX = (clientX - rect.left) * scaleX;
            const svgY = (clientY - rect.top) * scaleY;
            const nx = (svgX - CAKE_CX) / CAKE_RX;
            const ny = (svgY - CAKE_CY) / CAKE_RY;
            return Math.atan2(ny, nx);
        }

        // Real life: you pick the knife up off the table first, THEN you cut.
        window.pickUpKnife = function() {
            if (!canDragCut()) {
                const zone = document.getElementById('cakeEmoji');
                if (zone && !cakeCut && candlesBlown < totalCandles) {
                    gsap.to(zone, { x:-8, duration:0.06, yoyo:true, repeat:5, onComplete:()=>{ gsap.set(zone,{x:0}); } });
                }
                return;
            }
            if (knifePickedUp) return;
            knifePickedUp = true;
            vibrate(15);

            const tableKnife = document.getElementById('tableKnife');
            const label = document.getElementById('knifeHolderLabel');
            if (tableKnife) tableKnife.classList.remove('ready');
            if (tableKnife) tableKnife.classList.add('picked');
            if (label) label.classList.remove('show');

            // The blade lifts off the table and hovers over the left edge of the cake, ready to slice
            const zone = document.getElementById('cakeEmoji');
            const knife = document.getElementById('knifeCursor');
            if (zone) zone.classList.add('knife-active'); // cake stops floating once the knife is in hand — feels grounded, like real cutting
            if (zone && knife) {
                const rect = zone.getBoundingClientRect();
                knife.style.setProperty('--knife-rot', '-35deg');
                knife.style.left = (rect.width * 0.1) + 'px';
                knife.style.top = (rect.height * 0.5) + 'px';
                knife.classList.add('active');
                gsap.fromTo(knife, { scale:1.3, opacity:0 }, { scale:1, opacity:1, duration:0.35, ease:'back.out' });
            }

            const statusDiv = document.getElementById('cakeStatus');
            const cutHint = document.getElementById('cutHint');
            if (statusDiv) statusDiv.textContent = '🔪 Now hold and drag across the cake to slice it';
            if (cutHint) cutHint.classList.add('ready');
        };

        function onCutPointerDown(e) {
            const zone = document.getElementById('cakeEmoji');
            if (!canDragCut()) {
                if (!cakeCut && candlesBlown < totalCandles) {
                    gsap.to(zone, { x:-8, duration:0.06, yoyo:true, repeat:5, onComplete:()=>{ gsap.set(zone,{x:0}); } });
                }
                return;
            }
            // Touching the cake before tapping the knife still works — it just picks the knife up for you first
            if (!knifePickedUp) pickUpKnife();
            cutDragActive = true;
            cutStartX = e.clientX;
            cutStartY = e.clientY;
            // Wherever the user first presses down sets the direction of the cut — free choice, not fixed
            cutAngle = pointerToCakeAngle(e.clientX, e.clientY);
            updateSliceLineAngle(cutAngle);
            document.getElementById('knifeCursor').classList.add('active');
            updateKnifePosition(e.clientX, e.clientY);
            if (zone.setPointerCapture) { try { zone.setPointerCapture(e.pointerId); } catch(err) {} }
        }

        function onCutPointerMove(e) {
            if (!cutDragActive) return;
            updateKnifePosition(e.clientX, e.clientY);

            // The cut keeps following the pointer — drag toward any part of the cake and that's
            // the direction (and portion) that gets sliced.
            cutAngle = pointerToCakeAngle(e.clientX, e.clientY);
            const lineLength = updateSliceLineAngle(cutAngle);

            const dragDist = Math.hypot(e.clientX - cutStartX, e.clientY - cutStartY);
            const rawProgress = Math.min(dragDist / CUT_DRAG_THRESHOLD, 1);
            // Cake has resistance: the blade catches a little at first and only "gives way"
            // as the cut goes deeper, instead of slicing through at a perfectly even rate.
            const progress = Math.pow(rawProgress, 1.35);

            // Tilt the blade to match the direction being cut, plus a small wobble that fades
            // out as the cut completes — like a hand sawing through something slightly firm
            // rather than gliding through empty air.
            const knife = document.getElementById('knifeCursor');
            if (knife) {
                const tiltDeg = (cutAngle * 180 / Math.PI) - 90;
                const wobble = Math.sin(dragDist / 7) * (7 * (1 - progress));
                knife.style.setProperty('--knife-rot', (tiltDeg + wobble) + 'deg');
            }

            const line = document.getElementById('sliceLine');
            const shadow = document.getElementById('sliceShadow');
            // A touch of resistance jitter on the cut line itself, so the edge doesn't reveal
            // in a perfectly smooth sweep — settles down as the slice nears completion.
            const jitter = (Math.random() - 0.5) * 3 * (1 - progress);
            const offset = String(Math.max(0, lineLength - lineLength * progress + jitter));
            if (line) line.style.strokeDashoffset = offset;
            if (shadow) shadow.style.strokeDashoffset = offset;

            // A tiny buzz as the knife actually moves through the cake (no sound, just haptics),
            // pulsing a bit more insistently while there's still resistance left in the cut
            if (dragDist > 4 && Math.random() < 0.35 + 0.4 * (1 - progress)) vibrate(4);

            if (rawProgress >= 1) {
                cutDragActive = false;
                cutCake();
            }
        }

        function onCutPointerUp() {
            if (!cutDragActive) return;
            cutDragActive = false;
            // The knife is already in hand — it stays hovering over the cake rather than vanishing,
            // just like you wouldn't drop the knife after a half-finished cut in real life.
            if (!cakeCut) resetSliceLine();
        }

        document.getElementById('cakeEmoji').addEventListener('pointerdown', onCutPointerDown);
        document.getElementById('cakeEmoji').addEventListener('pointermove', onCutPointerMove);
        window.addEventListener('pointerup', onCutPointerUp);
        window.addEventListener('pointercancel', onCutPointerUp);

        // Real envelope-opening beat: seal breaks, flap unfolds, then the letter slides up into view.
        function resetEnvelope() {
            const envelope = document.getElementById('envelopeEl');
            const flap = document.getElementById('envelopeFlap');
            const seal = document.getElementById('waxSeal');
            const scene = document.getElementById('envelopeScene');
            const letter = document.getElementById('letterContainer');
            if (!envelope) return;
            envelope.classList.remove('opened');
            flap.classList.remove('open');
            seal.classList.remove('breaking');
            scene.classList.remove('hidden');
            gsap.set(scene, { clearProps:'opacity' });
            letter.classList.remove('show');
            gsap.set(letter, { opacity:0, y:40 });
            resetLetterTyping();
        }

        window.openEnvelope = function() {
            const envelope = document.getElementById('envelopeEl');
            const flap = document.getElementById('envelopeFlap');
            const seal = document.getElementById('waxSeal');
            const scene = document.getElementById('envelopeScene');
            const letter = document.getElementById('letterContainer');
            if (!envelope || envelope.classList.contains('opened')) return;
            envelope.classList.add('opened');
            vibrate([10, 30]);

            // The wax seal cracks and falls away first
            seal.classList.add('breaking');
            setTimeout(() => {
                // Then the flap folds open, like unsealing a real envelope
                flap.classList.add('open');
                vibrate(15);
            }, 280);

            // Finally the letter slides up out of the envelope and the envelope fades out
            setTimeout(() => {
                letter.classList.add('show');
                gsap.fromTo(letter, { opacity:0, y:50, scale:0.96 }, { opacity:1, y:0, scale:1, duration:0.7, ease:'power2.out' });
                gsap.to(scene, { opacity:0, duration:0.5, onComplete:() => scene.classList.add('hidden') });
                setTimeout(typeLetter, 400);
            }, 900);
        };
